如何启用云端共享？
1. 前往 supabase.com 免费注册并创建项目
2. 在 SQL Editor 中执行下方建表语句
3. 将项目 URL 与 anon key 填入下方
4. 点击「连接云端」，数据即可多设备共享
create table if not exists dcsmp_data ( id int primary key, data jsonb not null default '{"ocs":[]}'::jsonb, updated_at timestamptz default now() ); insert into dcsmp_data (id,data) values (1,'{"ocs":[]}'::jsonb) on conflict (id) do nothing; alter table dcsmp_data enable row level security; create policy "allow_all" on dcsmp_data for all using (true) with check (true);
